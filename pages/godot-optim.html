CollisionResolver — Physique bille-bille

GDScript — collision_resolver.gd (simplifié)
# Plan de layers
# Layer 1 : Joueur      détecte 2,3,6,7
# Layer 2 : Ennemis     détecte 3,4,5,7
# Layer 3 : Murs/géo
# Layer 7 : Zones terrain (Area3D)

func resolve(a: Node, b: Node) -> void:
    var normal = (b.global_position - a.global_position).normalized()
    var rel_vel = a.velocity - b.velocity
    var impulse = rel_vel.dot(normal)
    if impulse > 0: return  # déjà en séparation
    var ma = a.get_mass(); var mb = b.get_mass()
    var j = -((1.0 + restitution) * impulse) / (1.0/ma + 1.0/mb)
    a.apply_resolved_velocity(a.velocity - normal * (j/ma))
    b.apply_resolved_velocity(b.velocity + normal * (j/mb))

SpatialHash — queries O(1)

GDScript — spatial_hash.gd (sous-système PhysicsManager)
const CELL_SIZE := 200.0
var _grid: Dictionary = {}

func _physics_process(_delta: float) -> void:
    _grid.clear()
    for entity in get_tree().get_nodes_in_group("spatial_entities"):
        if not is_instance_valid(entity): continue
        var cell = _to_cell(entity.global_position)
        if not _grid.has(cell): _grid[cell] = []
        _grid[cell].append(entity)

func query_radius(pos: Vector3, radius: float, group: String = "") -> Array:
    var result = []; var cr = ceili(radius / CELL_SIZE)
    var cc = _to_cell(pos); var r2 = radius * radius
    for dx in range(-cr, cr+1):
        for dz in range(-cr, cr+1):
            var cell = cc + Vector3i(dx, 0, dz)
            if not _grid.has(cell): continue
            for e in _grid[cell]:
                if group != "" and not e.is_in_group(group): continue
                if pos.distance_squared_to(e.global_position) <= r2:
                    result.append(e)
    return result

IA tickée séparément

GDScript — ai_tick_scheduler.gd (sous-système EnemyManager)
# Taux de tick par type d'ennemi :
# Faibles : 0.15s (~7fps)  |  Normaux : 0.10s (~10fps)
# Puissants : 0.08s (~12fps) | Mini-boss : 0.05s (~20fps)
# Boss / Trône Renversé : 0.033s (~30fps)

func _physics_process(delta: float) -> void:
    for enemy in _active_enemies:
        enemy._ai_timer -= delta
        if enemy._ai_timer <= 0:
            enemy._ai_timer = enemy.data.ai_tick_rate
            enemy.ai_component.tick()
