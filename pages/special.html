Cas spéciaux

4 patterns pour les effets qui ne rentrent pas dans le pipeline synchrone standard. Dans tous les cas, SkillManager reste le seul point de coordination.

Règle : les cas spéciaux n'inventent pas de nouveaux canaux de communication. Ils utilisent des mécanismes déjà présents dans SkillManager.

Vue d'ensemble des 4 patterns

Pattern 1 — Observateur enregistré (Symbiote)

GDScript — symbiote_node.gd
extends Node

var power_ratio: float = 0.5
var duration_remaining: float = 12.0
var skill_manager: Node

# Appelé directement par SkillManager — pas de signal
func on_effect_executed(effect_data: EffectData, ctx: Dictionary) -> void:
    if ctx.get("is_symbiote_copy", false): return  # anti-boucle
    var nearest = ctx["physics_manager"].query_radius(global_position, 200.0, "enemies")
    if nearest.is_empty(): return
    var sym_ctx = ctx.duplicate()
    sym_ctx["target"]           = nearest[0]
    sym_ctx["is_symbiote_copy"] = true
    var sym_data = effect_data.duplicate()
    sym_data.base_damage *= power_ratio
    EffectRunner.execute(sym_data, sym_ctx)

func _process(delta: float) -> void:
    duration_remaining -= delta
    if duration_remaining <= 0.0:
        skill_manager.unregister_active_observer(self)
        queue_free()

Pattern 2 — Sous-système conditionnel (Boucle Temporelle)

GDScript — temporal_buffer.gd (sous-système SkillManager)
extends Node
var target: Node
var positions: Array = []
const BUFFER_DURATION = 3.0

func _physics_process(_delta: float) -> void:
    var now = Time.get_ticks_msec() / 1000.0
    positions.append({"pos": target.global_position, "time": now})
    positions = positions.filter(func(e): return now - e["time"] <= BUFFER_DURATION)

func get_position_n_seconds_ago(n: float) -> Vector3:
    var target_time = Time.get_ticks_msec() / 1000.0 - n
    for entry in positions:
        if entry["time"] >= target_time: return entry["pos"]
    return positions[0]["pos"] if not positions.is_empty() else target.global_position

Pattern 3 — Pipeline alternatif (Roulette du Destin)

GDScript — pick_and_execute.gd
extends EffectBehaviour

func apply(ctx: Dictionary, data: EffectData) -> Dictionary:
    var pool: Array[EffectData] = data.get("effect_pool", [])
    if pool.is_empty(): return ctx
    EffectRunner.execute(pool[randi() % pool.size()], ctx)
    ctx["pipeline_stopped"] = true
    return ctx

Pattern 4 — Pipeline asynchrone (Tempête d'Éclairs)

GDScript — SkillManager.schedule_repeated_effect()
func schedule_repeated_effect(
        effect: EffectData, ctx: Dictionary,
        count: int, delay: float) -> void:

    var executed := 0
    get_tree().create_timer(delay).timeout.connect(func():
        var targets = PhysicsManager.query_radius(ctx["position"], 300.0, "enemies")
        if not targets.is_empty():
            var rctx = ctx.duplicate()
            rctx["target"] = targets[randi() % targets.size()]
            EffectRunner.execute(effect, rctx)
        executed += 1
        if executed < count:
            schedule_repeated_effect(effect, ctx, count - executed, delay)
    )
